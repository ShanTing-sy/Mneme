# Mneme V1.0.0 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 SillyTavern 交付一个挂在魔法棒里的自用记忆插件 Mneme，实现卷宗式记忆、羁绊、道具与折叠剧情线。

**Architecture:** 真·酒馆扩展，Vue 3 + Vite + Zod。`core/` 放零宿主依赖的纯函数，`store/` 管聊天状态与全局设置，`host/` 负责酒馆事件与注入接线，`ui/` 只做面板展示与编辑。所有酒馆能力通过全局 `SillyTavern.getContext()` 获取，不导入 `@sillytavern/*`。

**Tech Stack:** Vue 3.5、Vite 7、Zod 4、TypeScript、Node 24 内置测试运行器。

- [ ] **Step 5: 创建 `src/core/migrate.ts`**

```ts
import type { MnemeState } from './types.ts';
import { createEmptyState, normalizeState } from './schema.ts';

export type MigrationStatus = 'empty' | 'ok' | 'repaired' | 'failed';

export interface MigrationResult {
  state: MnemeState;
  status: MigrationStatus;
}

export function migrateState(raw: unknown): MigrationResult {
  if (!raw || typeof raw !== 'object') {
    return { state: createEmptyState(), status: 'empty' };
  }

  const obj = raw as Record<string, unknown>;

  if (obj.version === 1) {
    const state = normalizeState(obj);
    return state
      ? { state, status: 'ok' }
      : { state: createEmptyState(), status: 'failed' };
  }

  const patched = { ...obj, version: 1 };
  const state = normalizeState(patched);
  return state
    ? { state, status: 'repaired' }
    : { state: createEmptyState(), status: 'failed' };
}
```

```ts
import type { Delta, Importance } from './types.ts';

const BLOCK_RE = /<!--\s*MNEME([\s\S]*?)MNEME\s*-->/i;
const FENCE_RE = /```(?:mneme|json)\s*([\s\S]*?)```/i;

export interface ParseResult {
  ok: boolean;
  delta?: Delta;
  block?: string;
  inner?: string;
  reason?: string;
}

export function parseDelta(text: string): ParseResult {
  if (!text || typeof text !== 'string') {
    return { ok: false, reason: 'empty' };
  }

  let inner: string | undefined;
  let block: string | undefined;

  const blockMatch = text.match(BLOCK_RE);
  if (blockMatch) {
    block = blockMatch[0];
    inner = blockMatch[1];
  } else {
    const fenceMatch = text.match(FENCE_RE);
    if (fenceMatch && /"summary"\s*:/.test(fenceMatch[1])) {
      block = fenceMatch[0];
      inner = fenceMatch[1];
    }
  }

  if (!inner) {
    return { ok: false, reason: 'no-block' };
  }

  const jsonText = extractObject(cleanJson(inner));
  if (!jsonText) {
    return { ok: false, reason: 'no-json', block, inner };
  }

  try {
    const raw = JSON.parse(jsonText);
    const delta = normalizeDelta(raw);
    return delta
      ? { ok: true, delta, block, inner }
      : { ok: false, reason: 'invalid', block, inner };
  } catch {
    return { ok: false, reason: 'bad-json', block, inner };
  }
}

function extractObject(text: string): string | null {
  const start = text.indexOf('{');
  const end = text.lastIndexOf('}');
  return start >= 0 && end > start ? text.slice(start, end + 1) : null;
}

function cleanJson(text: string): string {
  let out = text.replace(/^\uFEFF/, '').trim();
  out = out.replace(/[\u201C\u201D]/g, '"').replace(/[\u2018\u2019]/g, "'");
  out = out.replace(/,\s*([}\]])/g, '$1');
  return out;
}

function asString(value: unknown): string {
  return typeof value === 'string' ? value.trim() : '';
}

function normalizeDelta(raw: unknown): Delta | null {
  if (!raw || typeof raw !== 'object') return null;
  const obj = raw as Record<string, any>;

  const summary = asString(obj.summary);
  if (!summary) return null;

  const importance: Importance = ['normal', 'major', 'permanent'].includes(obj.importance)
    ? obj.importance
    : 'normal';

  return {
    summary,
    storyTime: asString(obj.storyTime),
    location: asString(obj.location),
    characters: Array.isArray(obj.characters)
      ? obj.characters.filter((x: unknown) => typeof x === 'string').slice(0, 12)
      : [],
    result: asString(obj.result),
    importance,
    anchor: asString(obj.anchor) || undefined,
    bonds: Array.isArray(obj.bonds)
      ? obj.bonds.filter((b: any) => b && typeof b.name === 'string')
      : [],
    items: Array.isArray(obj.items)
      ? obj.items.filter((i: any) => i && typeof i.name === 'string')
      : [],
    plot: obj.plot && typeof obj.plot === 'object' ? obj.plot : undefined,
    seal: obj.seal && typeof obj.seal === 'object' ? obj.seal : undefined,
  };
}
```

```ts
  let removed = false;

    removed = true;

  return { text: output, removed: true };
}
```

```ts
export function hashText(text: string): string {
  let hash = 0x811c9dc5;
  for (let i = 0; i < text.length; i += 1) {
    hash ^= text.charCodeAt(i);
    hash = Math.imul(hash, 0x01000193) >>> 0;
  }
  return `${hash.toString(36)}:${text.length.toString(36)}`;
}
```

```ts
import type { MnemeState } from './types.ts';
import { hashText } from './hash.ts';
import { cloneData } from './clone.ts';

export interface ChatLike {
  mes: string;
  is_user?: boolean;
}

export interface ReconcileResult {
  state: MnemeState;
  dropped: string[];
  remapped: number;
}

export function removeEntryLinks(state: MnemeState, entryIds: string[]): void {
  const set = new Set(entryIds);
  for (const bond of Object.values(state.bonds)) {
    bond.events = bond.events.filter((event) => !event.entryId || !set.has(event.entryId));
  }
  state.items = state.items.filter((item) => !item.entryId || !set.has(item.entryId));
}

export function reconcileState(state: MnemeState, chat: ChatLike[]): ReconcileResult {
  const buckets = new Map<string, number[]>();

  chat.forEach((message, index) => {
    if (!message || message.is_user) return;
    const hash = hashText(String(message.mes ?? ''));
    const list = buckets.get(hash) ?? [];
    list.push(index);
    buckets.set(hash, list);
  });

  const used = new Set<number>();
  const next = cloneData(state);
  const dropped: string[] = [];
  let remapped = 0;

  for (const entry of next.entries) {
    const candidates = buckets.get(entry.mesHash);
    const index = candidates?.find((candidate) => !used.has(candidate));
    if (index === undefined) {
      dropped.push(entry.id);
      continue;
    }
    used.add(index);
    if (index !== entry.messageId) {
      entry.messageId = index;
      remapped += 1;
    }
  }

  if (dropped.length > 0) {
    next.entries = next.entries.filter((entry) => !dropped.includes(entry.id));
    removeEntryLinks(next, dropped);
  }

  return { state: next, dropped, remapped };
}
```

```ts
    const node = timeline.nodes.find((item) => item.id === id);

  }

    const node = timeline.nodes.find((item) => item.id === plot.current);
    if (node) {
      node.status = node.status === '已发生' ? '已发生' : '进行中';
      timeline.currentId = node.id;

export function currentNode(timeline: Timeline): TimelineNode | null {
  if (timeline.nodes.length === 0) return null;
  if (timeline.currentId) {
    return timeline.nodes.find((node) => node.id === timeline.currentId) ?? null;
  }
  return (
    timeline.nodes.find((node) => node.status === '进行中') ??
    timeline.nodes.find((node) => node.status === '未发生') ??
    timeline.nodes[0]
  );
}

export function visibleNodes(timeline: Timeline, view: DirectorView): TimelineNode[] {
  const current = currentNode(timeline);
  if (!current) return [];
  if (view === 'full') return timeline.nodes;

  const index = timeline.nodes.findIndex((node) => node.id === current.id);
  if (index < 0) return [current];

  if (view === 'current') return [current];
  if (view === 'current-next') return timeline.nodes.slice(index, index + 2);
  return timeline.nodes.slice(index, index + 3);
}

export function buildDirectorText(timeline: Timeline | null, view: DirectorView): string {
  if (!timeline) return '';
  const current = currentNode(timeline);
  if (!current) return '';
```

```ts
import type {
  Bond,
  Delta,
  DeltaSeal,
  MemoryEntry,
  MnemeState,
  Volume,
} from './types.ts';
import { applyPlot } from './timeline.ts';
import { removeEntryLinks } from './reconcile.ts';
import { clamp } from './bonds.ts';
import { sealVolume } from './seal.ts';
import { cloneData } from './clone.ts';

export interface DeltaMeta {
  messageId: number;
  swipeId: number;
  mesHash: string;
  now?: number;
  volumeSize?: number;
}

export interface ApplyResult {
  state: MnemeState;
  entry: MemoryEntry;
  sealed: Volume | null;
  warnings: string[];
}

function ensureBond(state: MnemeState, name: string, now: number): Bond {
  if (!state.bonds[name]) {
    state.bonds[name] = {
      name,
      initialScore: 0,
      manualDelta: 0,
      impression: '',
      events: [],
      updatedAt: now,
    };
  }
  return state.bonds[name];
}

export function applyDelta(state: MnemeState, delta: Delta, meta: DeltaMeta): ApplyResult {
  const now = meta.now ?? Date.now();
  const next = cloneData(state);
  const warnings: string[] = [];

  // 用「消息索引 + swipe 序号」定位旧条目，id 本身带时间戳以避免与归档条目撞号
  const previous = next.entries.find(
    (entry) => entry.messageId === meta.messageId && entry.swipeId === meta.swipeId,
  );
  const entryId = previous?.id ?? `m${meta.messageId}-s${meta.swipeId}-${now.toString(36)}`;

  next.entries = next.entries.filter((entry) => entry.id !== entryId);
  removeEntryLinks(next, [entryId]);

  const entry: MemoryEntry = {
    id: entryId,
    messageId: meta.messageId,
    swipeId: meta.swipeId,
    mesHash: meta.mesHash,
    at: now,
    storyTime: delta.storyTime ?? '',
    location: delta.location ?? '',
    summary: (delta.summary ?? '').trim() || '（模型未提供摘要）',
    characters: delta.characters ?? [],
    result: delta.result ?? '',
    importance: delta.importance ?? 'normal',
    anchor: delta.anchor,
    volume: next.stats.sealedVolumes + 1,
    source: 'model',
  };

  next.entries.push(entry);
  next.stats.totalEntries += 1;

  for (const item of delta.bonds ?? []) {
    const name = (item.name ?? '').trim();
    if (!name) {
      warnings.push('bonds 缺少 name，已跳过');
      continue;
    }
    const bond = ensureBond(next, name, now);
    const deltaValue = clamp(Math.round(Number(item.delta) || 0), -20, 20);
    if (deltaValue !== 0 || item.event) {
      bond.events.push({
        id: `e${now.toString(36)}-${bond.events.length}`,
        text: (item.event || '互动').slice(0, 200),
        delta: deltaValue,
        importance: item.importance ?? entry.importance,
        at: now,
        entryId,
        messageId: meta.messageId,
      });
      bond.updatedAt = now;
    }
    if (item.impression) {
      bond.impression = item.impression.slice(0, 300);
      bond.updatedAt = now;
    }
  }

  for (const item of delta.items ?? []) {
    const name = (item.name ?? '').trim();
    if (!name) {
      warnings.push('items 缺少 name，已跳过');
      continue;
    }
    const existing = next.items.find((candidate) => candidate.name === name);

    if (item.action === 'gain') {
      if (existing) {
        existing.status = 'held';
        existing.source = item.source || existing.source;
        existing.effect = item.effect || existing.effect;
        existing.entryId = entryId;
        existing.messageId = meta.messageId;
      } else {
        next.items.push({
          id: `i${now.toString(36)}-${next.items.length}`,
          name,
          source: item.source ?? '未知',
          effect: item.effect ?? '',
          status: 'held',
          acquiredAt: now,
          entryId,
          messageId: meta.messageId,
          note: item.note,
        });
      }
      continue;
    }

    if (!existing) {
      warnings.push(`道具 ${name} 未找到，${item.action} 已忽略`);
      continue;
    }

    if (item.action === 'lose') existing.status = 'lost';
    if (item.action === 'consume') existing.status = 'consumed';
    if (item.action === 'change') {
      existing.effect = item.effect || existing.effect;
      existing.note = item.note ?? existing.note;
    }
    existing.entryId = entryId;
    existing.messageId = meta.messageId;
  }

  if (delta.plot && next.timeline) {
    applyPlot(next.timeline, delta.plot);
  }

  const size = clamp(Math.round(meta.volumeSize ?? 10), 5, 30);
  let sealed: Volume | null = null;
  if (next.entries.length >= size) {
    sealed = sealVolume(next, delta.seal as DeltaSeal | undefined, size);
  }

  return { state: next, entry, sealed, warnings };
}

export function removeEntry(state: MnemeState, entryId: string): MnemeState {

  next.entries = next.entries.filter((entry) => entry.id !== entryId);
  for (const volume of next.volumes) {
    volume.entries = volume.entries.filter((entry) => entry.id !== entryId);
  }
  removeEntryLinks(next, [entryId]);
```

```ts
import type { MemoryEntry, MnemeSettings, MnemeState } from './types.ts';
import { bondScore, bondTier } from './bonds.ts';
import { buildDirectorText } from './timeline.ts';

export function estimateTokens(text: string): number {
  let cjk = 0;
  let other = 0;
  for (const char of text) {
    if (/[\u3400-\u9FFF\uF900-\uFAFF\u3000-\u303F\uFF00-\uFFEF]/.test(char)) {
      cjk += 1;
    } else {
      other += 1;
    }
  }
  return Math.ceil(cjk + other / 4);
}

function renderEntryLine(entry: MemoryEntry): string {
  const meta = [entry.storyTime, entry.location].filter(Boolean).join('·');
  const mark = entry.importance === 'permanent' ? '★' : entry.importance === 'major' ? '◆' : '';
  const anchor = entry.anchor ? `（锚点：${entry.anchor}）` : '';
  return `${mark}${meta ? `[${meta}] ` : ''}${entry.summary}${anchor}`;
}

function renderEntries(entries: MemoryEntry[], title: string): string {
  if (entries.length === 0) return '';
  return `<Mneme·${title}>\n${entries.map(renderEntryLine).join('\n')}\n</Mneme·${title}>`;
}

function buildInstruction(state: MnemeState, settings: MnemeSettings): string {
  const size = settings.volumeSize;
  const count = state.entries.length + 1;
  const needSeal = count >= size;
  const plotLine = state.timeline
    ? '- plot：仅在剧情节点推进时填写，advance 为已完成的节点 id，current 为进入的节点 id。\n'
    : '';

  return [
    '<Mneme·记录要求>',
    '在正文结束后，另起一行输出一个 HTML 注释块，且只输出这一个块：',
    '<!--MNEME',
    '{"summary":"…","storyTime":"…","location":"…","characters":["…"],"result":"…","importance":"normal|major|permanent","bonds":[],"items":[]}',
    'MNEME-->',
    '- summary：完整流畅的一段话，客观记录本轮发生了什么（谁、在哪、做了什么、结果），25–100 字。',
    '- importance：仅当出现“第一次”“人格改变”“强烈情绪事件”时才用 major 或 permanent；其余一律 normal。',
    '- anchor：仅 major/permanent 填写，写一个具体的感知细节。',
    '- bonds：本轮好感变化，delta 为 -20~20 的整数；event 写触发变化的事；只有印象变化时才写 impression。',
    '- items：action 为 gain/lose/consume/change。',
    plotLine + (needSeal
      ? `- 本条是本卷第 ${size} 条，必须同时输出 "seal":{"title":"卷名","recap":"150-200字复盘"}。`
      : `- 当前卷已有 ${count - 1}/${size} 条。`),
    '不要复述这段说明，不要把它写进正文。',
    '</Mneme·记录要求>',
  ].join('\n');
}

interface Section {
  key: string;
  text: string;
  priority: number;
  required: boolean;
}

export function buildInjection(
  state: MnemeState | null,
  settings: MnemeSettings,
  _deps: { userName: string; charName: string },
): string {
  if (!state) return '';

  const permanent: MemoryEntry[] = [];
  const recent: MemoryEntry[] = [];
  for (const entry of state.entries) {
    if (entry.importance === 'permanent') permanent.push(entry);
    else recent.push(entry);
  }

  const archivedPermanent: MemoryEntry[] = [];
  const archivedMajor: MemoryEntry[] = [];
  const firstMajorVolume = state.stats.sealedVolumes - settings.majorRetentionVolumes + 1;
  for (const volume of state.volumes) {
    for (const entry of volume.entries) {
      if (entry.importance === 'permanent') archivedPermanent.push(entry);
      else if (entry.importance === 'major' && volume.index >= firstMajorVolume) archivedMajor.push(entry);
    }
  }

  const sections: Section[] = [
    {
      key: 'director',
      text: buildDirectorText(state.timeline, settings.directorView),
      priority: 90,
      required: false,
    },
    {
      key: 'permanent',
      text: renderEntries([...permanent, ...archivedPermanent], '永久记忆'),
      priority: 95,
      required: true,
    },
  ];

  recent.forEach((entry, index) => {
    sections.push({
      key: `recent-${entry.id}`,
      text: renderEntryLine(entry),
      priority: 50 + index,
      required: false,
    });
  });

  sections.push({
    key: 'major',
    text: renderEntries(archivedMajor, '重要记忆'),
    priority: 70,
    required: false,
  });

  const recaps = state.volumes
    .slice(-3)
    .map((volume) => `第${volume.index}卷·${volume.title}：${volume.recap}`)
    .join('\n');
  sections.push({
    key: 'volumes',
    text: recaps ? `<Mneme·卷宗>\n${recaps}\n</Mneme·卷宗>` : '',
    priority: 40,
    required: false,
  });

  const bondLines = Object.values(state.bonds).map((bond) => {
    const score = bondScore(bond);
    const impression = bond.impressionOverride ?? bond.impression;
    return `${bond.name}｜好感 ${score}（${bondTier(score).label}）｜印象：${impression || '暂无'}`;
  });
  sections.push({
    key: 'bonds',
    text: bondLines.length ? `<Mneme·羁绊>\n${bondLines.join('\n')}\n</Mneme·羁绊>` : '',
    priority: 80,
    required: false,
  });

  const itemLines = state.items
    .filter((item) => item.status === 'held')
    .map((item) => `${item.name}（来源：${item.source || '未知'}；作用：${item.effect || '未知'}）`);
  sections.push({
    key: 'items',
    text: itemLines.length ? `<Mneme·道具>\n${itemLines.join('\n')}\n</Mneme·道具>` : '',
    priority: 45,
    required: false,
  });

  sections.push({
    key: 'instruction',
    text: buildInstruction(state, settings),
    priority: 999,
    required: true,
  });

  const active = sections.filter((section) => section.text);
  const dropOrder = active
    .filter((section) => !section.required)
    .sort((a, b) => a.priority - b.priority);

  const total = () => estimateTokens(active.filter((section) => section.text).map((section) => section.text).join('\n'));

  let used = total();
  for (const section of dropOrder) {
    if (used <= settings.tokenBudget) break;
    section.text = '';
    used = total();
  }

  const byKey = new Map(active.map((section) => [section.key, section]));
  const recentText = active
    .filter((section) => section.key.startsWith('recent-') && section.text)
    .map((section) => section.text)
    .join('\n');

  return [
    byKey.get('director')?.text,
    byKey.get('permanent')?.text,
    recentText ? `<Mneme·近期记忆>\n${recentText}\n</Mneme·近期记忆>` : '',
    byKey.get('major')?.text,
    byKey.get('volumes')?.text,
    byKey.get('bonds')?.text,
    byKey.get('items')?.text,
    byKey.get('instruction')?.text,
  ]
    .filter(Boolean)
    .join('\n');
}
```

```ts
export function loadFromChat(): MnemeState | null {
  const ctx = getCtx();
  if (!ctx || !ctx.chatMetadata) {
    store.state = null;
    store.status = 'no-chat';
    return null;
  }

  const raw = ctx.chatMetadata.mneme;

  if (result.status === 'failed' && raw) {

  store.state = result.state;
  store.status = result.status;

}

export function commitState(next: MnemeState): void {
  store.state = next;
  const ctx = getCtx();
  if (!ctx || !ctx.chatMetadata) return;

  ctx.chatMetadata.mneme = JSON.parse(JSON.stringify(next));

  if (saveTimer) clearTimeout(saveTimer);
  saveTimer = setTimeout(() => {
    saveTimer = null;
    try {
      void ctx.saveMetadata?.();
    } catch (error) {
      console.warn('[Mneme] save failed', error);
    }
  }, 400);
}

export async function flushState(): Promise<void> {
  if (!saveTimer) return;
  clearTimeout(saveTimer);
  saveTimer = null;
  const ctx = getCtx();
  try {
    await ctx?.saveMetadata?.();
  } catch (error) {

  }
```

```ts
export function mergeSettings(base: MnemeSettings, patch: Partial<MnemeSettings> | undefined): MnemeSettings {
  const merged: MnemeSettings = {

    ...(patch ?? {}),
    ui: { ...base.ui, ...((patch?.ui as Partial<MnemeSettings['ui']>) ?? {}) },
  };
  merged.volumeSize = Math.min(30, Math.max(5, Math.round(Number(merged.volumeSize) || 10)));

}

export function readSettings(): MnemeSettings {

  if (!ext) return mergeSettings(DEFAULTS, undefined);

  if (!ext.mneme || typeof ext.mneme !== 'object') ext.mneme = {};

  try {
    ctx.saveSettingsDebounced?.();

  }
}
```

```ts
import type { Delta } from '../core/types.ts';
import { parseDelta } from '../core/parse.ts';
import { stripDelta } from '../core/strip.ts';
import { hashText } from '../core/hash.ts';
import { applyDelta } from '../core/reduce.ts';
import { reconcileState } from '../core/reconcile.ts';
import { buildInjection } from '../core/inject.ts';
import { getCtx } from './context.ts';
import { store, loadFromChat, commitState, flushState } from '../store/chat-state.ts';
import { readSettings } from '../store/ui-settings.ts';
import { panelState } from '../ui/panel-state.ts';
import { setWandDot } from './wand.ts';

let registered = false;

export function initHost(retries = 40): void {
  const ctx = getCtx();
  if (!ctx) {
    if (retries > 0) setTimeout(() => initHost(retries - 1), 250);
    return;
  }
  if (registered) return;
  registered = true;

  const events = ctx.eventTypes;
  ctx.eventSource.on(events.MESSAGE_RECEIVED, (id: number, type: string) => {
    void onMessageReceived(id, type);
  });
  ctx.eventSource.on(events.MESSAGE_EDITED, (id: number) => {
    void onMessageEdited(id);
  });
  ctx.eventSource.on(events.MESSAGE_DELETED, () => {
    void onMessageDeleted();
  });
  ctx.eventSource.on(events.CHAT_CHANGED, () => {
    void onChatChanged();
  });
  ctx.eventSource.on(events.GENERATION_STARTED, () => {
    refreshInjection();
  });

  loadFromChat();
  refreshInjection();
}

export function refreshInjection(): void {
  const ctx = getCtx();
  if (!ctx) return;
  const settings = readSettings();

  if (!settings.enabled || !store.state) {
    try {
      ctx.setExtensionPrompt('mneme', '', 0, 0, false, 0);
    } catch (error) {
      console.warn('[Mneme] clear injection failed', error);
    }
    return;
  }

  const text = buildInjection(store.state, settings, {
    userName: ctx.name1 || 'user',
    charName: ctx.characters?.[ctx.characterId]?.name || 'char',
  });

  try {
    ctx.setExtensionPrompt('mneme', text, 0, 0, false, 0);
  } catch (error) {
    console.warn('[Mneme] inject failed', error);
  }
}

export async function onMessageReceived(messageId: number, _type: string): Promise<void> {
  const ctx = getCtx();
  if (!ctx || !store.state) return;

  const settings = readSettings();
  if (!settings.enabled) return;

  const message = ctx.chat?.[messageId];
  if (!message || message.is_user) return;

  const original = String(message.mes ?? '');
  const parsed = parseDelta(original);

  if (!parsed.ok && !parsed.block) {
    if (settings.showMissingBadge) {
      panelState.missing += 1;
      setWandDot(true);
    }
    return;
  }

  const stripped = stripDelta(original);
  const cleanText = stripped.text;
  const swipeId = Number.isInteger(message.swipe_id) ? message.swipe_id : 0;

  const delta: Delta = parsed.ok && parsed.delta
    ? parsed.delta
    : {
        summary: `（原始记录）${String(parsed.inner ?? '').slice(0, 400)}`,
        importance: 'normal',
      };

  const result = applyDelta(store.state, delta, {
    messageId,
    swipeId,
    mesHash: hashText(cleanText),
    volumeSize: settings.volumeSize,
  });

  if (!parsed.ok) {
    result.entry.source = 'raw';
    panelState.raw += 1;
    setWandDot(true);
  }

  message.mes = cleanText;
  if (Array.isArray(message.swipes) && message.swipes[swipeId] !== undefined) {
    message.swipes[swipeId] = cleanText;
  }

  try {
    ctx.updateMessageBlock?.(messageId, message, { rerenderMessage: true });
  } catch (error) {
    console.warn('[Mneme] rerender failed', error);
  }

  if (result.sealed) {
    panelState.lastSealed = result.sealed.index;
    setWandDot(true);
  }

  commitState(result.state);
  refreshInjection();
  await flushState();
}

export async function onMessageEdited(messageId: number): Promise<void> {
  const ctx = getCtx();
  if (!ctx || !store.state) return;

  const message = ctx.chat?.[messageId];
  if (!message) return;

  const settings = readSettings();
  const parsed = parseDelta(String(message.mes ?? ''));
  if (!parsed.ok || !parsed.delta) return;

  const stripped = stripDelta(String(message.mes ?? ''));
  const swipeId = Number.isInteger(message.swipe_id) ? message.swipe_id : 0;
  const result = applyDelta(store.state, parsed.delta, {
    messageId,
    swipeId,
    mesHash: hashText(stripped.text),
    volumeSize: settings.volumeSize,
  });

  message.mes = stripped.text;
  try {
    ctx.updateMessageBlock?.(messageId, message, { rerenderMessage: true });
  } catch (error) {
    console.warn('[Mneme] rerender failed', error);
  }

  commitState(result.state);
  refreshInjection();
  await flushState();
}

export async function onMessageDeleted(): Promise<void> {
  const ctx = getCtx();
  if (!ctx || !store.state) return;

  const result = reconcileState(store.state, ctx.chat ?? []);
  if (result.dropped.length === 0 && result.remapped === 0) return;

  commitState(result.state);
  refreshInjection();
  await flushState();
}

export async function onChatChanged(): Promise<void> {
  loadFromChat();
  refreshInjection();
}
```

```ts
import type { Bond, MnemeState, Timeline, TimelineNode } from '../core/types.ts';
import { clamp } from '../core/bonds.ts';
import { cloneData } from '../core/clone.ts';
import { getCtx } from './context.ts';
import { store, commitState, flushState } from '../store/chat-state.ts';
import { readSettings } from '../store/ui-settings.ts';
import { panelState } from '../ui/panel-state.ts';
import { refreshInjection } from './events.ts';

export interface CharacterContext {
  card: string;
  world: string;
  charName: string;
  worldBooks: string[];
}

const PARSE_SCHEMA = {
  type: 'object',
  properties: {
    bonds: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          name: { type: 'string' },
          initialScore: { type: 'integer' },
          impression: { type: 'string' },
          events: { type: 'array', items: { type: 'string' } },
        },
        required: ['name', 'initialScore', 'impression'],
      },
    },
    timeline: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          title: { type: 'string' },
          summary: { type: 'string' },
          type: { type: 'string', enum: ['主线', '支线', '伏笔', '里程碑'] },
          keywords: { type: 'array', items: { type: 'string' } },
        },
        required: ['title', 'summary'],
      },
    },
  },
  required: ['bonds'],
};

let running = false;

export function scheduleInitParse(): void {
  const settings = readSettings();
  if (!settings.autoInitBonds && !settings.autoTimeline) return;
  setTimeout(() => {
    void runInitParse({ bonds: settings.autoInitBonds, timeline: settings.autoTimeline });
  }, 1800);
}

export async function runInitParse(options: { bonds: boolean; timeline: boolean; force?: boolean }): Promise<void> {
  const ctx = getCtx();
  if (!ctx || !store.state || running) return;

  const state = store.state;
  const needBonds = options.bonds && (options.force || Object.keys(state.bonds).length === 0);
  const needTimeline = options.timeline && (options.force || !state.timeline);
  if (!needBonds && !needTimeline) return;

  running = true;
  panelState.initRunning = true;
  panelState.initError = '';

  try {
    const context = await collectCharacterContext();
    const prompt = buildParsePrompt(context, { bonds: needBonds, timeline: needTimeline });

    const raw = await ctx.generateRaw({
      systemPrompt: '你是剧情分析助手。只输出 JSON，不要输出任何解释或代码围栏。',
      prompt,
      responseLength: 1200,
      jsonSchema: PARSE_SCHEMA,
    });

    const parsed = JSON.parse(extractJson(String(raw ?? '')));
    const safe = needTimeline ? parsed : { bonds: parsed?.bonds };
    const next = applyParsed(store.state, safe, {
      char: context.charName,
      worldBooks: context.worldBooks,
    });

    commitState(next);
    refreshInjection();
    await flushState();
  } catch (error: any) {
    panelState.initError = error?.message || String(error);
    console.warn('[Mneme] init parse failed', error);
  } finally {
    running = false;
    panelState.initRunning = false;
  }
}

export function extractJson(text: string): string {
  const fenced = text.match(/```(?:json)?\s*([\s\S]*?)```/i);
  const source = fenced ? fenced[1] : text;
  const start = source.indexOf('{');
  const end = source.lastIndexOf('}');
  if (start < 0 || end <= start) return source.trim();
  return source.slice(start, end + 1).trim();
}

export function applyParsed(
  state: MnemeState,
  parsed: any,
  sources: { char: string; worldBooks: string[] },
): MnemeState {
  const next = cloneData(state);
  const now = Date.now();

  for (const item of Array.isArray(parsed?.bonds) ? parsed.bonds : []) {
    const name = typeof item?.name === 'string' ? item.name.trim() : '';
    if (!name || next.bonds[name]) continue;

    const bond: Bond = {
      name,
      initialScore: clamp(Math.round(Number(item.initialScore) || 0), 0, 100),
      manualDelta: 0,
      impression: typeof item.impression === 'string' ? item.impression.slice(0, 300) : '',
      events: Array.isArray(item.events)
        ? item.events
            .filter((text: unknown) => typeof text === 'string' && text.trim())
            .map((text: string, index: number) => ({
              id: `init-${now.toString(36)}-${index}`,
              text: text.slice(0, 200),
              delta: 0,
              importance: 'major' as const,
              at: now,
              entryId: null,
              messageId: null,
            }))
        : [],
      updatedAt: now,
    };
    next.bonds[name] = bond;
  }

  if (Array.isArray(parsed?.timeline) && parsed.timeline.length > 0) {
    const nodes: TimelineNode[] = parsed.timeline.map((item: any, index: number) => ({
      id: `n${index + 1}`,
      order: index,
      title: typeof item?.title === 'string' ? item.title : `节点 ${index + 1}`,
      summary: typeof item?.summary === 'string' ? item.summary : '',
      type: ['主线', '支线', '伏笔', '里程碑'].includes(item?.type) ? item.type : '主线',
      keywords: Array.isArray(item?.keywords)
        ? item.keywords.filter((key: unknown) => typeof key === 'string')
        : [],
      status: '未发生',
      spoiler: true,
    }));

    const timeline: Timeline = {
      generatedAt: now,
      sources: { char: sources.char, worldBooks: sources.worldBooks },
      nodes,
      currentId: nodes[0]?.id ?? null,
    };
    next.timeline = timeline;
    next.config.worldBooks = sources.worldBooks;
  }

  return next;
}

export async function collectCharacterContext(): Promise<CharacterContext> {
  const ctx = getCtx();
  const chid = ctx?.characterId;
  const character = ctx?.characters?.[chid];
  const fields = ctx?.getCharacterCardFields?.({ chid }) ?? {};
  const charName = character?.name ?? '角色';

  const card = [
    `姓名：${charName}`,
    fields.description ? `描述：${fields.description}` : '',
    fields.personality ? `性格：${fields.personality}` : '',
    fields.scenario ? `场景：${fields.scenario}` : '',
    fields.firstMessage ? `开场白：${fields.firstMessage}` : '',
    fields.system ? `系统提示：${fields.system}` : '',
    fields.jailbreak ? `后置指令：${fields.jailbreak}` : '',
  ].filter(Boolean).join('\n');

  const configured = store.state?.config.worldBooks ?? [];
  const characterWorld = character?.data?.extensions?.world;
  const globalSelected = ctx?.extensionSettings?.world_info?.globalSelect;
  const worldBooks: string[] = configured.length > 0
    ? configured
    : [characterWorld, ...(Array.isArray(globalSelected) ? globalSelected : [])].filter(Boolean);

  const chunks: string[] = [];
  for (const name of worldBooks) {
    try {
      const book = await ctx?.loadWorldInfo?.(name);
      const entries = book?.entries ?? {};
      for (const key of Object.keys(entries)) {
        const entry = entries[key];
        if (!entry || entry.disable) continue;
        const content = String(entry.content ?? '').trim();
        if (!content) continue;
        chunks.push(`【${name}·${entry.comment || key}】\n${content.slice(0, 1200)}`);
        if (chunks.join('\n').length > 12000) break;
      }
    } catch (error) {
      console.warn('[Mneme] world book load failed', name, error);
    }
    if (chunks.join('\n').length > 12000) break;
  }

  return { card: card.slice(0, 8000), world: chunks.join('\n').slice(0, 12000), charName, worldBooks };
}

function buildParsePrompt(context: CharacterContext, need: { bonds: boolean; timeline: boolean }): string {
  return [
    '【角色卡】',
    context.card,
    '',
    '【世界书摘录】',
    context.world || '（无）',
    '',
    '【任务】',
    need.bonds ? '1. 依据角色卡与世界书中已有的关系，推断每个主要角色对 user 的初始好感度（0=完全陌生，100=生死之交），并写出一句角色对 user 的当前印象。' : '',
    need.bonds ? '2. 列出角色卡/世界书中已经发生的、对关系有影响的事件（只写已经发生的）。' : '',
    need.timeline ? '3. 梳理一条 6-12 个节点的剧情时间轴，按可能发生的先后排序，涵盖主线、支线、伏笔、里程碑。节点要具体，但不要写死结局。' : '',
    '只输出 JSON，不要输出解释或代码围栏。',
  ].filter(Boolean).join('\n');
```
